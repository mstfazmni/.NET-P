CREATE TABLE Athletes (

	Id INT PRIMARY KEY IDENTITY(1,1)
	Name NAVCHAR(100) NOT NULL,
	WeightClass NAVCHAR(50),
	Wins INT DEFAULT 0,
	Losses INT DEFAULT 0
);


CREATE TABLE TrainingPlans (
	Id INT PRIMATY KEY IDENTITY(1,1),
	AthleteId INT NOT NULL,
	PlanName NAVCHAR(100),
	FOREIGN KEY (AthleteId) REFERENCES Athlete(Id)
)
